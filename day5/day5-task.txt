 Use $match to show sales with quantity above 10.
db.sales.aggregate([
  {
    $match: {
      quantity: { $gt: 10 }
    }
  }
])
 Use $addFields to add a new field revenue = quantity * price.
db.sales.aggregate([
  {
    $addFields: {
      revenue: {
        $multiply: ["$quantity", "$price"]
      }
    }
  }
])
 Use $group to find total and average revenue for each category.
db.sales.aggregate([
  {
    $addFields: {
      revenue: {
        $multiply: ["$quantity", "$price"]
      }
    }
  },
  {
    $group: {
      _id: "$category",
      totalRevenue: { $sum: "$revenue" },
      averageRevenue: { $avg: "$revenue" }
    }
  }
])

 Use $bucket to group items by price range.
db.sales.aggregate([
  {
    $bucket: {
      groupBy: "$price",
      boundaries: [0, 100, 1000, 10000, 100000],
      default: "Other",
      output: {
        totalItems: { $sum: 1 }
      }
    }
  }
])
 Use $lookup to join with a new suppliers collection.
db.createCollection("suppliers")

db.suppliers.insertMany([
  {
    category: "Electronics",
    supplierName: "Tech Suppliers"
  },
  {
    category: "Furniture",
    supplierName: "Home Suppliers"
  },
  {
    category: "Stationery",
    supplierName: "Office Suppliers"
  }
])


db.sales.aggregate([
  {
    $lookup: {
      from: "suppliers",
      localField: "category",
      foreignField: "category",
      as: "supplierDetails"
    }
  }
])
